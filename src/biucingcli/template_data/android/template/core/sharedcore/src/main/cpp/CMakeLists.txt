cmake_minimum_required(VERSION 3.22)
project(BiucingShared LANGUAGES C CXX)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../../shared/core" portable-core)
add_library(biucing_shared SHARED jni.cpp)
target_link_libraries(biucing_shared PRIVATE Biucing::Core)
target_compile_features(biucing_shared PRIVATE cxx_std_20)
target_compile_options(biucing_shared PRIVATE -fvisibility=hidden)
if(ANDROID)
    target_link_options(biucing_shared PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
    find_package(JNI REQUIRED)
    target_include_directories(biucing_shared PRIVATE ${JNI_INCLUDE_DIRS})
endif()
